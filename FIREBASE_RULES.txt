rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() {
      return request.auth != null;
    }

    function isAdmin() {
      return signedIn() && request.auth.token.email == "yogesh.bokhoree@gmail.com";
    }

    match /wcUsers/{userId} {
      allow create: if signedIn() && request.auth.uid == userId;
      allow read, update: if isAdmin() || (signedIn() && request.auth.uid == userId);
      allow delete: if isAdmin();
    }

    match /wcCards/{cardId} {
      allow create: if signedIn() && request.resource.data.userId == request.auth.uid;
      allow read: if isAdmin() || (signedIn() && resource.data.userId == request.auth.uid);
      allow update, delete: if isAdmin();
    }

    match /wcOdds/{matchId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }

    match /wcBadges/{badgeId} {
      allow read: if signedIn();
      allow create, update, delete: if isAdmin();
    }
  }
}
