rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function isAdmin() { return signedIn() && request.auth.token.email == "yogesh.bokhoree@gmail.com"; }

    match /users/{userId} {
      allow create: if signedIn() && request.auth.uid == userId;
      allow read, update: if isAdmin() || (signedIn() && request.auth.uid == userId);
    }

    match /portfolios/{id} {
      allow create: if signedIn() && request.resource.data.userId == request.auth.uid;
      allow read: if isAdmin() || (signedIn() && resource.data.userId == request.auth.uid);
    }

    match /footballPortfolios/{id} {
      allow create: if signedIn() && request.resource.data.userId == request.auth.uid;
      allow read: if isAdmin() || (signedIn() && resource.data.userId == request.auth.uid);
    }
  }
}
